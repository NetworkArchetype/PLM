A.1 Core Model — plm_formalized/model.py

The core mathematical engine of the PLM framework. All arithmetic uses Python’s decimal.Decimal at 80-digit precision to guarantee exact reproducibility.

from __future__ import annotations
from dataclasses import dataclass
from decimal import Decimal, getcontext

getcontext().prec = 80


def hex_to_int(hex_str: str) -> int:
    """Convert a hexadecimal string to an integer."""
    s = hex_str.strip().lower()
    if s.startswith("0x"):
        s = s[2:]
    if not all(c in "0123456789abcdef" for c in s):
        raise ValueError(f"Invalid hex string: {hex_str}")
    return int(s, 16)


@dataclass(frozen=True)
class PLMInputs:
    """Immutable container for all PLM transform parameters."""
    pi: Decimal
    lam: Decimal
    mu: Decimal
    x: int
    public_hash_hex: str
    block_size: int
    crc_decimal: int

    def y(self) -> int:
        return hex_to_int(self.public_hash_hex)

    def c(self) -> int:
        c = self.block_size + self.crc_decimal
        if c <= 0:
            raise ValueError("C = block_size + crc_decimal must be positive")
        return c


def plm_ratio(pi: Decimal, lam: Decimal, mu: Decimal) -> Decimal:
    """Compute R = (pi * lambda) / mu."""
    if mu == 0:
        raise ZeroDivisionError("mu cannot be zero")
    return (pi * lam) / mu


def plm_secret_value(inputs: PLMInputs) -> Decimal:
    """Compute S = ((pi * Y) * (lambda * X)) / (mu * C)."""
    if inputs.mu == 0:
        raise ZeroDivisionError("mu cannot be zero")
    Y = Decimal(inputs.y())
    C = Decimal(inputs.c())
    numerator = (inputs.pi * Y) * (inputs.lam * Decimal(inputs.x))
    denominator = inputs.mu * C
    return numerator / denominator

A.2 Stateful Machine — plm_formalized/stateful.py

The stateful time-series wrapper that evolves PLM inputs over discrete time steps using an injected advance function (Strategy pattern).

from __future__ import annotations
from dataclasses import dataclass
from typing import Callable
from .model import PLMInputs, plm_secret_value

@dataclass(frozen=True)
class PLMState:
    """Immutable snapshot of the PLM state machine at time t."""
    t: int
    inputs: PLMInputs

AdvanceFn = Callable[[PLMState], PLMState]


class StatefulPLM:
    """Deterministic state machine that evolves PLMInputs over time."""

    def __init__(self, initial: PLMState, advance: AdvanceFn):
        self._state = initial
        self._advance = advance

    @property
    def state(self) -> PLMState:
        return self._state

    def step(self) -> PLMState:
        """Advance one time step and return the new state."""
        self._state = self._advance(self._state)
        return self._state

    def value(self) -> Decimal:
        """Compute S for the current state."""
        return plm_secret_value(self._state.inputs)

A.3 Quantum Temporal — plm_formalized/quantum_temporal.py

Encodes PLM time-series outputs into parametric quantum circuits using Google Cirq.

from __future__ import annotations
import math
from dataclasses import dataclass
from .model import plm_secret_value
from .stateful import StatefulPLM

@dataclass
class QuantumTemporalConfig:
    shots: int = 200
    scale: float = 1.0


def simulate_time_series(machine: StatefulPLM, steps: int,
                          cfg: QuantumTemporalConfig) -> list:
    import cirq
    results = []
    q = cirq.LineQubit(0)

    for i in range(steps):
        S = float(machine.value())
        theta = (S * cfg.scale) % (2 * math.pi)

        circuit = cirq.Circuit(
            cirq.H(q),
            cirq.rz(theta).on(q),
            cirq.H(q),
            cirq.measure(q, key='m'),
        )

        sim = cirq.Simulator()
        result = sim.run(circuit, repetitions=cfg.shots)
        m = result.measurements['m'].reshape(-1)
        p1 = float(m.mean())
        expZ = 1.0 - 2.0 * p1

        results.append({
            't': machine.state.t,
            'S': str(machine.value()),
            'theta': theta,
            'p1': p1,
            'expZ': expZ,
        })
        machine.step()

    return results

A.4 Ruby Implementation — Code_core_3/plm-formalized/ruby/plm.rb

The canonical Ruby mirror using BigDecimal for arbitrary-precision arithmetic. Structurally equivalent to the Python implementation.

# frozen_string_literal: true
require "bigdecimal"
require "bigdecimal/util"

module PLM
  module_function

  def hex_to_int(hex_str)
    s = hex_str.strip.downcase
    s = s[2..] if s.start_with?("0x")
    raise ArgumentError, "invalid hex" unless s.match?(/\A[0-9a-f]+\z/)
    s.to_i(16)
  end

  def plm_ratio(pi:, lam:, mu:)
    mu = BigDecimal(mu.to_s)
    raise ZeroDivisionError, "mu cannot be 0" if mu.zero?
    BigDecimal(pi.to_s) * BigDecimal(lam.to_s) / mu
  end

  def secret_value(pi:, lam:, mu:, x:, public_hash_hex:,
                    block_size:, crc_decimal:)
    pi  = BigDecimal(pi.to_s)
    lam = BigDecimal(lam.to_s)
    mu  = BigDecimal(mu.to_s)
    raise ZeroDivisionError, "mu cannot be 0" if mu.zero?

    y = hex_to_int(public_hash_hex)
    c = Integer(block_size) + Integer(crc_decimal)
    raise ArgumentError, "C must be positive" if c <= 0

    numerator   = (pi * BigDecimal(y.to_s)) *
                  (lam * BigDecimal(x.to_s))
    denominator = mu * BigDecimal(c.to_s)
    numerator / denominator
  end
end

A.5 Test Suite — tests/test_model.py

The complete unit test suite verifying core PLM functions.

from decimal import Decimal
import pytest
from plm_formalized import (
    PLMInputs, hex_to_int, plm_ratio, plm_secret_value,
)


def test_hex_to_int():
    assert hex_to_int("0x0a") == 10
    assert hex_to_int("ff") == 255


def test_hex_to_int_invalid():
    with pytest.raises(Exception):
        hex_to_int("not-hex")


def test_plm_ratio():
    pi  = Decimal("3.14")
    lam = Decimal("2")
    mu  = Decimal("4")
    assert plm_ratio(pi, lam, mu) == (pi * lam) / mu


def test_plm_secret_value_deterministic():
    inp = PLMInputs(
        pi=Decimal("3.141592653589793"),
        lam=Decimal("1.618033988749894"),
        mu=Decimal("1"),
        x=123,
        public_hash_hex="a3f1c9",
        block_size=4096,
        crc_decimal=987654321,
    )
    s1 = plm_secret_value(inp)
    s2 = plm_secret_value(inp)
    assert s1 == s2


def test_c_must_be_positive():
    inp = PLMInputs(
        pi=Decimal("3.14"), lam=Decimal("1"),
        mu=Decimal("1"), x=1,
        public_hash_hex="0a",
        block_size=0, crc_decimal=0,
    )
    with pytest.raises(ValueError):
        inp.c()

Remark C.1 — Transcendence Conjecture

\(R = \pi\varphi/e\) is a product of two transcendental numbers (\(\pi\), \(e\)) and one algebraic irrational (\(\varphi\)). By heuristic arguments from transcendental number theory, \(R\) is almost certainly transcendental, though a rigorous proof would require tools such as Schanuel’s conjecture or extensions of the Lindemann–Weierstrass theorem. The irrationality of \(R\) is certain, since a rational ratio of \(\pi\varphi\) to \(e\) would imply an algebraic relation among these constants that is not known to exist.
